# 测试指南

> YiPet Chrome 扩展测试完整指南：Vitest 单元测试、Content Script 测试、Service Worker 测试、API 层测试、E2E 测试策略、测试环境配置、常见问题。

## 一、测试架构

### 测试分层

```
E2E 测试（Cypress/Playwright + Chrome Extension）
  ├── 完整用户流程：注入 → 聊天 → 配置切换
  └── 跨世界通信端到端验证

集成测试
  ├── Content Script + Service Worker 通信
  ├── ApiClient + YiAi 后端
  └── Popup + chrome.storage 读写

单元测试（Vitest）
  ├── Chat Store（Pinia）状态管理
  ├── ApiClient 重试和错误处理
  ├── IPC Relay 消息验证
  ├── CDN Injector 资源加载
  ├── 工具函数和纯逻辑
  └── 类型校验
```

### 测试金字塔

```
        /\
       /E2E\         少量，覆盖核心流程
      /------\
     /集成测试\       中等，覆盖跨模块交互
    /----------\
   / 单元测试   \     大量，覆盖所有纯逻辑
  /--------------\
```

```typescript
// tests/mocks/chrome.ts
import { vi } from "vitest";

export const chromeStorageMock = (() => {
  let store: Record<string, unknown> = {};

  return {
    local: {
      get: vi.fn((keys: string | string[] | null, callback?: (result: Record<string, unknown>) => void) => {
        const result: Record<string, unknown> = {};
        if (keys === null) {
          Object.assign(result, store);
        } else if (Array.isArray(keys)) {
          for (const key of keys) {
            if (key in store) result[key] = store[key];
          }
        } else if (typeof keys === "string") {
          result[keys] = store[keys];
        }
        if (callback) callback(result);
        return Promise.resolve(result);
      }),
      set: vi.fn((items: Record<string, unknown>, callback?: () => void) => {
        Object.assign(store, items);
        if (callback) callback();
        return Promise.resolve();
      }),
      remove: vi.fn((keys: string | string[], callback?: () => void) => {
        const keyList = Array.isArray(keys) ? keys : [keys];
        for (const key of keyList) {
          delete store[key];
        }
        if (callback) callback();
        return Promise.resolve();
      }),
      clear: vi.fn(() => {
        store = {};
        return Promise.resolve();
      }),
      getBytesInUse: vi.fn((callback?: (bytes: number) => void) => {

        return Promise.resolve(bytes);

      get: vi.fn(),
      set: vi.fn(),

    },
  };
})();
```

### Chat Store 测试

```typescript
// tests/chat/stores/chat.test.ts
import { describe, it, expect, beforeEach, vi } from "vitest";
import { setActivePinia, createPinia } from "pinia";
import { useChatStore } from "../../src/chat/stores/chat";
import type { ApiServices } from "../../src/api";

describe("useChatStore", () => {
  let store: ReturnType<typeof useChatStore>;
  let mockApi: ApiServices;

  beforeEach(() => {
    setActivePinia(createPinia());

    mockApi = {
      chat: { sendMessage: vi.fn(), streamChat: vi.fn() },
      session: {
        getByKey: vi.fn(),
        upsert: vi.fn(),
        delete: vi.fn(),
        list: vi.fn(),
      },
      data: {
        queryDocuments: vi.fn(),
        createDocument: vi.fn(),
        updateDocument: vi.fn(),
        deleteDocument: vi.fn(),
      },
      knowledge: { readEntry: vi.fn(), writeEntry: vi.fn(), listTree: vi.fn() },
      rag: { streamChat: vi.fn(), streamFileChat: vi.fn(), search: vi.fn() },
      file: { readFile: vi.fn(), writeFile: vi.fn(), upload: vi.fn() },
    } as unknown as ApiServices;

    // 注入 API services
    vi.stubGlobal("__YIPET_API_SERVICES__", mockApi);

    store = useChatStore();
  });

  describe("状态管理", () => {
    it("初始状态应为默认值", () => {
      expect(store.messages).toEqual([]);
      expect(store.isProcessing).toBe(false);
      expect(store.sessions).toEqual([]);
      expect(store.currentSessionId).toBeNull();
    });

    it("状态变更应自动响应", () => {
      store.$patch({ chatVisible: true });
      expect(store.chatVisible).toBe(true);
    });

    it("$reset 应恢复初始状态", () => {
      store.$patch({ chatVisible: true, messages: [{ role: "user", content: "hi" }] });
      store.$reset();
      expect(store.chatVisible).toBe(false);
      expect(store.messages).toEqual([]);
    });
  });

  describe("消息发送", () => {
    it("isProcessing 时拒绝发送", async () => {
      store.isProcessing = true;
      await store.sendMessage("hello");
      expect(mockApi.chat.streamChat).not.toHaveBeenCalled();
    });

    it("空消息拒绝发送", async () => {
      await store.sendMessage("");
      await store.sendMessage("   ");
      expect(mockApi.chat.streamChat).not.toHaveBeenCalled();
    });

    it("发送消息后 isProcessing 为 true", async () => {
      mockApi.chat.streamChat = vi.fn(async function* () {
        yield { type: "text", content: "hello" };
      });
      const promise = store.sendMessage("hello");
      expect(store.isProcessing).toBe(true);
      await promise;
    });
  });

  describe("会话管理", () => {
    it("createSession 应创建新会话并设为当前", async () => {
      const id = await store.createSession();
      expect(store.currentSessionId).toBe(id);
      expect(store.sessions.length).toBe(1);
      expect(store.sessions[0].id).toBe(id);
    });

    it("deleteSession 删除当前会话应切换到最近会话", async () => {
      const id1 = await store.createSession();
      const id2 = await store.createSession();
      await store.deleteSession(id2);
      expect(store.currentSessionId).toBe(id1);
    });
  });
});
```

### ApiClient 测试

```typescript
// tests/api/client.test.ts
import { describe, it, expect, beforeEach, vi } from "vitest";
import { ApiClient } from "../../src/api/client";

describe("ApiClient", () => {
  let client: ApiClient;

  beforeEach(() => {
    client = new ApiClient("http://localhost:10086");
    globalThis.fetch = vi.fn();
  });

  describe("request", () => {
    it("应组装正确的 RPC 信封", async () => {
      vi.mocked(fetch).mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve({ code: 0, message: "ok", data: { list: [], total: 0 } }),
      } as Response);

      await client.request({
        moduleName: "services.database.data_service",
        methodName: "query_documents",
        parameters: { cname: "projects", filter: { status: "active" } },
      });

      expect(fetch).toHaveBeenCalledWith(
        "http://localhost:10086/",
        expect.objectContaining({
          method: "POST",
          body: JSON.stringify({
            module_name: "services.database.data_service",
            method_name: "query_documents",
            parameters: { cname: "projects", filter: { status: "active" } },
          }),
        })
      );
    });

    it("HTTP 错误应返回 ok=false 的 ApiResponse", async () => {
      vi.mocked(fetch).mockResolvedValueOnce({
        ok: false,
        status: 500,
        statusText: "Internal Server Error",
        headers: { get: () => "application/json" },
        json: () => Promise.resolve({ detail: "Internal Server Error" }),
      } as unknown as Response);

      const result = await client.rpc("test", "test", {});
      expect(result.ok).toBe(false);
      expect(result.status).toBe(500);
      expect(result.error).toBeDefined();
    });

    it("业务错误码应返回 ok=false 的 ApiResponse", async () => {
      vi.mocked(fetch).mockResolvedValueOnce({
        ok: true,
        status: 200,
        headers: { get: () => "application/json" },
        json: () => Promise.resolve({ code: 401, message: "Token 过期", data: null }),
      } as unknown as Response);

      const result = await client.rpc("test", "test", {});
      expect(result.ok).toBe(false);
      expect(result.error).toBe("Token 过期");
    });

    it("网络错误应返回 ok=false 的 ApiResponse", async () => {
      vi.mocked(fetch).mockRejectedValueOnce(new TypeError("Failed to fetch"));

      const result = await client.rpc("test", "test", {});
      expect(result.ok).toBe(false);
      expect(result.status).toBe(0);
      expect(result.error).toContain("Failed to fetch");
    });

    it("AbortError 不重试，直接返回错误", async () => {
      vi.mocked(fetch).mockRejectedValueOnce(
        new DOMException("Aborted", "AbortError")
      );

      const result = await client.rpc("test", "test", {});
      expect(result.ok).toBe(false);
      expect(result.error).toContain("timed out or was aborted");
    });
  });
});
```

### IPC Relay 测试

```typescript
// tests/content/ipc/relay.test.ts
import { describe, it, expect, beforeEach, vi } from "vitest";

describe("IPC Relay", () => {
  const IPC_SECRET = "test-secret-123";
  let dispatchSecureEvent: (name: string, detail: unknown) => void;

  beforeEach(() => {
    vi.clearAllMocks();
    dispatchSecureEvent = (name, detail) => {
      window.dispatchEvent(new CustomEvent(name, {
        detail: {
          __yipet: true,
          __signature: IPC_SECRET,
          __timestamp: Date.now(),
          data: detail,
        },
      }));
    };
  });

  describe("dispatchSecureEvent 安全验证", () => {
    it("应拒绝缺少 __yipet 标记的事件", () => {
      const handler = vi.fn();
      window.addEventListener('yipet:roleChanged', handler as EventListener);

      window.dispatchEvent(new CustomEvent('yipet:roleChanged', {
        detail: { __signature: IPC_SECRET, __timestamp: Date.now(), data: {} },
      }));

      expect(handler).not.toHaveBeenCalled();
    });

    it("应拒绝签名不匹配的事件", () => {
      const handler = vi.fn();
      window.addEventListener('yipet:roleChanged', handler as EventListener);

      window.dispatchEvent(new CustomEvent('yipet:roleChanged', {
        detail: { __yipet: true, __signature: 'wrong-secret', __timestamp: Date.now(), data: {} },
      }));

      expect(handler).not.toHaveBeenCalled();
    });

    it("应拒绝过期事件（5 秒过期）", () => {
      const handler = vi.fn();
      window.addEventListener('yipet:roleChanged', handler as EventListener);

      window.dispatchEvent(new CustomEvent('yipet:roleChanged', {
        detail: { __yipet: true, __signature: IPC_SECRET, __timestamp: Date.now() - 6000, data: {} },
      }));

      expect(handler).not.toHaveBeenCalled();
    });

    it("应接受合法事件", () => {
      const handler = vi.fn();
      window.addEventListener('yipet:roleChanged', handler as EventListener);

      dispatchSecureEvent('yipet:roleChanged', { role: 'cat' });

      expect(handler).toHaveBeenCalled();
    });
  });

  describe("chrome.runtime.onMessage 处理", () => {
    it("应处理 setRole action", () => {
      const sendResponse = vi.fn();
      const listener = vi.fn();

      chrome.runtime.onMessage.addListener(listener);
      // 模拟 Popup 发送 setRole
      listener({ action: 'setRole', role: 'cat' }, {}, sendResponse);

      expect(sendResponse).toHaveBeenCalledWith(
        expect.objectContaining({ success: true })
      );
    });
  });
});
```

## 十、测试环境差异与限制

### 10.1 jsdom vs 真实浏览器

| 功能 | jsdom | Chrome 真实环境 | 测试策略 |
|------|-------|---------------|----------|
| DOM 操作 | ✓ 支持 | ✓ 支持 | jsdom 单测 |
| `chrome.*` API | 需 mock | ✓ 原生 | 使用 mock（见第二节） |
| `window.postMessage` | ✓ 支持 | ✓ 支持 | jsdom 单测 |
| `MutationObserver` | ✓ 支持 | ✓ 支持 | jsdom 单测 |
| CSS 渲染 | ✗ 不支持 | ✓ 支持 | Playwright E2E |
| Service Worker | ✗ 不支持 | ✓ 支持 | Playwright E2E |
| Extension Context | ✗ 不支持 | ✓ 支持 | 手动测试 + Playwright |
| CSP 限制 | 不模拟 | 强制 | 构建验证 |

### 10.2 不可在 jsdom 中测试的场景

以下场景必须使用 Playwright 或在真实 Chrome 中手动测试：
- Service Worker 注册和生命周期
- Content Script 注入和隔离
- CSP 合规性验证
- chrome.storage 真实的读写性能
- 扩展图标和弹窗渲染
- 键盘快捷键冲突检测
