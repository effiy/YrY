# 测试策略指南

> **读完你将能够**：YiAi 各层测试策略：Mock 模式、异步测试、覆盖率目标、常见测试模式

> YiAi 测试的完整策略：分层测试方法、Mock 模式、异步测试、覆盖率目标。

## 一、测试分层

```
tests/
├── conftest.py                 # 共享 fixtures（mock_db, mock_repository）
├── shared/                     # 共享层测试（纯函数，无外部依赖）
│   ├── test_error_codes.py     # ErrorCode 枚举
│   ├── test_response.py        # StandardResponse 格式
│   └── test_utils.py           # 工具函数
├── data/                       # 数据层测试（Mock MongoDB）
│   └── test_repository.py      # _build_filter, 分页逻辑
├── domain/                     # 领域层测试（Mock data 层）
│   ├── test_auth.py            # JWT + bcrypt 逻辑
│   └── test_files.py           # 文件读写逻辑
├── services/                   # 服务层测试（Mock domain 层）
│   └── test_data_service.py    # 参数校验 + 业务编排
└── server/routes/              # 路由层测试（httpx AsyncClient）
    └── test_routes.py          # 端到端 HTTP 测试
```

### 分层原则

| 层 | Mock 什么 | 验证什么 |
|----|----------|---------|
| shared | 无 | 纯函数逻辑、错误码、响应格式 |
| data | MongoDB 连接 | `_build_filter` 查询构建、分页计算 |
| domain | `data/` 层 | 业务逻辑、异常抛出、边界情况 |
| services | `domain/` 层 | 参数校验、业务编排、错误传递 |
| routes | `services/` 层 | HTTP 状态码、响应格式、认证 |

## 二、共享 Fixtures（conftest.py）

```python
# tests/conftest.py
import pytest
import pytest_asyncio
from unittest.mock import AsyncMock, MagicMock, patch

@pytest.fixture
def mock_db():
    """Mock MongoDB 单例 — 所有 data 层测试使用"""
    db = AsyncMock()
    db.find_one = AsyncMock(return_value={"key": "test", "name": "Test"})
    db.find_many = AsyncMock(return_value=[{"key": "test"}])
    db.insert_one = AsyncMock(return_value="mocked_id")
    db.update_one = AsyncMock(return_value=True)
    db.delete_one = AsyncMock(return_value=True)
    db.count = AsyncMock(return_value=10)
    return db

@pytest.fixture
def mock_repository(mock_db):
    """Mock Repository — services 层测试使用"""
    from src.data.repository import Repository
    repo = Repository(mock_db)
    return repo

@pytest_asyncio.fixture
async def test_app():
    """创建测试用 FastAPI 应用"""
    from src.app import create_app
    return create_app()
```

## 三、关键 Mock 模式

### Mock MongoDB

```python
@pytest.mark.asyncio
async def test_find_one_returns_document(mock_db):
    """验证 find_one 正确返回文档"""
    mock_db.find_one = AsyncMock(return_value={"key": "proj-1", "name": "Test"})

    with patch("src.data.database.db", mock_db):
        from src.data.repository import query_documents
        result = await query_documents("projects", {"key": "proj-1"})

    assert result["list"][0]["name"] == "Test"
    mock_db.find_one.assert_called_once()

@pytest.mark.asyncio
async def test_find_one_not_found(mock_db):
    """验证文档不存在时返回 None"""
    mock_db.find_one = AsyncMock(return_value=None)

    with patch("src.data.database.db", mock_db):
        result = await db.find_one("projects", {"key": "nonexistent"})

    assert result is None
```

### Mock Ollama

```python
@pytest.mark.asyncio
async def test_chat_stream(mock_ollama):

    mock_client.chat = mock_stream
```

```python
async def test_external_api_timeout():
```

## 四、各层测试模式

### Shared 层（纯函数，最易测试）

```python
def test_error_code_values():
    """验证错误码枚举值正确"""
    assert ErrorCode.SUCCESS.code == 0
    assert ErrorCode.NOT_FOUND.code == 404
    assert ErrorCode.AI_UNAVAILABLE.code == 503

def test_standard_response_success():
    """验证成功响应格式"""
    result = StandardResponse.success({"key": "value"})
    assert result == {"code": 0, "message": "ok", "data": {"key": "value"}}

def test_standard_response_fail():
    """验证失败响应格式"""
    result = StandardResponse.fail(ErrorCode.NOT_FOUND, "项目不存在")
    assert result["code"] == 404
    assert result["data"] is None
```

### Domain 层（Mock data 层）

```python
@pytest.mark.asyncio
async def test_login_success(mock_db):
    """验证登录成功返回 Token"""
    import bcrypt
    hashed = bcrypt.hashpw("password".encode(), bcrypt.gensalt())
    mock_db.find_one = AsyncMock(return_value={
        "key": "admin", "password": hashed.decode(), "role": "admin"
    })

    with patch("src.domain.auth.auth.db", mock_db):
        with patch("src.domain.auth.auth.settings") as mock_settings:
            mock_settings.jwt_secret = "test-secret"
            mock_settings.jwt_expire_minutes = 1440

            result = await login("admin", "password")

    assert "token" in result
    assert result["user"]["key"] == "admin"
    assert "password" not in result["user"]

@pytest.mark.asyncio
async def test_login_wrong_password(mock_db):
    """验证密码错误抛出异常"""
    mock_db.find_one = AsyncMock(return_value={
        "key": "admin", "password": "hashed", "role": "admin"
    })

    with patch("src.domain.auth.auth.db", mock_db):
        with pytest.raises(BusinessException) as exc:
            await login("admin", "wrong")

    assert exc.value.error_code == ErrorCode.UNAUTHORIZED
```

### 路由层（httpx + ASGITransport）

```python
@pytest.mark.asyncio
async def test_health_endpoint(test_client):
    """验证健康检查端点"""
    response = await test_client.get("/health")
    assert response.status_code == 200
    data = response.json()
    assert data["code"] == 0
    assert data["data"]["status"] in ("ok", "degraded")

@pytest.mark.asyncio
async def test_rpc_endpoint_unauthorized(test_client):
    """验证未认证 RPC 调用"""
    response = await test_client.post(
        "/",
        json={
            "module_name": "services.database.data_service",
            "method_name": "query_documents",
            "parameters": {"cname": "projects"}
        }
    )
    assert response.status_code == 200
    assert response.json()["code"] != 0  # 业务错误码
```

## 五、覆盖率目标

| 模块 | 目标 | 理由 |
|------|------|------|
| `shared/` | >= 90% | 被所有层依赖，错误影响面大 |
| `data/` | >= 80% | 直接影响数据正确性 |
| `domain/` | >= 70% | 核心业务逻辑 |
| `services/` | >= 70% | 参数校验和异常路径 |
| `server/routes/` | >= 60% | HTTP 集成测试，重外部行为 |

## 六、运行测试

```bash
# 全量测试
python -m pytest tests/ -v

# 按层运行
pytest tests/shared/ -v           # 共享层（最快，< 1s）
pytest tests/data/ -v             # 数据层
pytest tests/domain/ -v           # 领域层
pytest tests/services/ -v         # 服务层
pytest tests/server/ -v           # 路由层（最慢，需启动 app）

# 单个测试
pytest tests/domain/test_auth.py::test_login_success -v

# 覆盖率
pytest tests/ --cov=src --cov-report=html --cov-report=term-missing

# 并行加速
pytest tests/ -n auto -v
```

## 七、测试编写检查清单

- [ ] 每个公共函数至少有 1 个 happy path 测试
- [ ] 每个 ErrorCode 抛出点有对应测试
- [ ] 分页参数边界（pageSize=0, pageSize=10000）有测试
- [ ] Mock 只在当前层以下（data 层 mock MongoDB，domain 层 mock data）
- [ ] 不使用真实 MongoDB/Ollama 连接（路由层集成测试除外）
- [ ] 异步函数用 `@pytest.mark.asyncio`
- [ ] 测试函数命名：`test_<功能>_<场景>_<预期>`
