# Pattern: Domain Service

> 领域服务完整模式：分层架构、Service 实现、Repository 注入、RPC 信封解包、响应包装、错误处理、依赖注入、测试策略、反模式。

## 概述

YiAi 使用 **分层架构 + Domain Service + Repository** 模式组织业务逻辑。严格的分层结构确保关注点分离：HTTP 层处理请求路由，Service 层处理业务编排，Domain 层处理核心逻辑，Data 层处理数据持久化。

### 导入规则

| 层 | 可导入 | 禁止导入 |
|----|--------|----------|
| HTTP (`server/`) | `shared/`, `services/` | `data/`, `domain/` |
| Service (`services/`) | `shared/`, `domain/` | `server/`, `data/` |
| Domain (`domain/`) | `shared/`, `data/` | `server/`, `services/` |
| Data (`data/`) | `shared/` | `server/`, `services/`, `domain/` |
| Shared (`shared/`) | — | 所有业务模块 |

## Service 实现模式

### 基本结构

```python
# src/services/database/data_service.py
from typing import Optional, Dict, Any
from src.domain.database.data import DataDomain
from src.shared.response import StandardResponse

class DataService:
    """通用 CRUD 服务 — 封装 domain 逻辑，供 routes 调用"""

    def __init__(self, domain: DataDomain):
        self._domain = domain

    async def query_documents(
        self,
        cname: str,
        filter: Optional[Dict[str, Any]] = None,
        page_num: int = 1,
        page_size: int = 20,
        order_by: Optional[str] = None,
        order_type: str = "desc",
        fields: Optional[list[str]] = None,
        exclude_fields: Optional[list[str]] = None,
    ) -> Dict[str, Any]:
        # 参数校验
        if page_num < 1:
            page_num = 1
        if page_size < 1 or page_size > 100:
            page_size = 20

        # 委托 domain 层执行
        return await self._domain.query_documents(
            cname=cname,
            filter=filter or {},
            page_num=page_num,
            page_size=page_size,
            order_by=order_by,
            order_type=order_type,
            fields=fields,
            exclude_fields=exclude_fields,
        )

    async def create_document(
        self, cname: str, data: Dict[str, Any]
    ) -> Dict[str, str]:
        # 参数校验
        if not cname or not data:
            raise ValueError("cname and data are required")

        # 委托 domain 层
        doc_id = await self._domain.create_document(cname, data)
        return {"inserted_id": doc_id}

    async def update_document(
        self, cname: str, key: Dict[str, Any], data: Dict[str, Any]
    ) -> Dict[str, int]:
        if not cname or not key:
            raise ValueError("cname and key are required")

        count = await self._domain.update_document(cname, key, data)
        return {"modified_count": count}

    async def delete_document(
        self, cname: str, key: Dict[str, Any]
    ) -> Dict[str, int]:
        if not cname or not key:
            raise ValueError("cname and key are required")

        count = await self._domain.delete_document(cname, key)
        return {"deleted_count": count}
```

### Domain 层实现

```python
# src/domain/database/data.py
from datetime import datetime
from src.data.repository import Repository

class DataDomain:
    """通用 CRUD 领域逻辑"""

    def __init__(self, repository: Repository):
        self._repo = repository

    async def query_documents(self, **kwargs) -> dict:
        return await self._repo.query(**kwargs)

    async def create_document(self, cname: str, data: dict) -> str:
        # 领域逻辑：添加时间戳
        now = datetime.utcnow()
        data.setdefault("created_at", now)
        data.setdefault("updated_at", now)
        return await self._repo.insert(cname, data)

    async def update_document(self, cname: str, key: dict, data: dict) -> int:
        data["updated_at"] = datetime.utcnow()
        return await self._repo.update(cname, key, {"$set": data})

    async def delete_document(self, cname: str, key: dict) -> int:
        return await self._repo.delete(cname, key)
```

## RPC 路由注册模式

### 路由定义

```python
@router.post("/")
async def rpc_dispatcher(
    request: Request,
    payload: RpcRequest,
):
    """RPC 分发器：根据 module_name.method_name 路由到对应 Service 方法"""
    module = payload.module_name
    method = payload.method_name
    params = payload.parameters or {}

    # 模块路由表
    handlers = {
        "services.database.data_service": _handle_data_service,
        "services.ai.chat_service": _handle_chat_service,
        "services.knowledge.knowledge_service": _handle_knowledge_service,
        "services.rag.rag_service": _handle_rag_service,
    }

    handler = handlers.get(module)
    if not handler:
        raise BusinessException(ErrorCode.UNKNOWN_MODULE, f"Unknown module: {module}")

    return await handler(request, method, params)

async def _handle_data_service(request: Request, method: str, params: dict):
    service = request.app.state.data_service

    method_map = {
        "query_documents": service.query_documents,
        "create_document": service.create_document,
        "update_document": service.update_document,
        "delete_document": service.delete_document,
    }

    handler = method_map.get(method)
    if not handler:
        raise BusinessException(ErrorCode.UNKNOWN_METHOD, f"Unknown method: {method}")

    # 注意：参数名转换（snake_case → 函数参数）
    # 前端传 module_name, method_name (snake_case)
    # 但 Python 函数参数是 snake_case，直接匹配
    result = await handler(**params)
    return StandardResponse.success(result)
```

### 响应包装

```python
# src/shared/response.py
from typing import Any, Generic, TypeVar

T = TypeVar("T")

class StandardResponse(BaseModel, Generic[T]):
    code: int = 0
    message: str = "ok"
    data: T | None = None

    @classmethod
    def success(cls, data: Any = None, message: str = "ok") -> dict:
        return {"code": 0, "message": message, "data": data}

    @classmethod
    def error(cls, code: int, message: str) -> dict:
        return {"code": code, "message": message, "data": None}
```

### 业务异常

```python
# src/shared/exceptions.py
class BusinessException(Exception):
    def __init__(self, code: ErrorCode | int, message: str = ""):
        self.code = int(code)
        self.message = message or ErrorCode(code).name if isinstance(code, ErrorCode) else ""

class AuthException(BusinessException):
    def __init__(self, message: str = "认证失败"):
        super().__init__(code=ErrorCode.UNAUTHORIZED, message=message)

class NotFoundException(BusinessException):
    def __init__(self, message: str = "资源不存在"):
        super().__init__(code=ErrorCode.NOT_FOUND, message=message)

class ValidationException(BusinessException):
    def __init__(self, message: str = "参数校验失败"):
        super().__init__(code=ErrorCode.INVALID_PARAM, message=message)
```

## 反模式

| 反模式 | 错误示例 | 正确做法 | 原因 |
|--------|----------|----------|------|
| 路由中直接操作 MongoDB | `@router.post("/data")` 中直接 `db.collection.find()` | 通过 Service → Domain → Repository 分层 | 业务逻辑散落在路由中，难以测试和复用 |
| Repository 中包含业务逻辑 | `Repository.query()` 中做参数校验 | 参数校验在 Service 层，Domain 处理领域逻辑，Repository 仅做数据访问 | 数据访问和业务规则耦合 |
| Service 直接使用 Motor | `self._db = AsyncIOMotorDatabase` | 通过 Domain → Repository 注入 | 无法替换数据源，难以单元测试 |
| 不包装响应 | 直接返回 dict | 使用 `StandardResponse.success()` 包装 | 响应格式不一致，前端解析困难 |
| 忽略参数校验 | 直接透传用户参数到数据库 | 校验 pageNum ≥ 1, pageSize ≤ 100 | 恶意请求可导致数据库压力 |
| 跨层依赖 | `domain/` 导入 `server/` | 严格遵循分层导入规则 | 循环依赖，架构腐化 |
| 异常处理不一致 | 有的抛 Exception，有的返回 error dict | 统一使用 BusinessException + ErrorCode | 错误处理混乱 |

## 约束

### 必须遵守
- 严格遵循分层架构：server → services → domain → data → shared
- Service 层处理参数校验和业务编排，不包含核心业务逻辑
- Domain 层处理核心业务逻辑，不导入 server/ 或处理 HTTP
- 所有 API 响应通过 `StandardResponse` 包装（`code: 0` 表示成功）
- 业务异常通过 `BusinessException` + `ErrorCode` 抛出
- 使用依赖注入（构造函数注入）而非全局单例
- 每层只依赖下一层和 shared 层

### 禁止
- 不在路由中直接操作 MongoDB
- 不在 Repository 中编写业务逻辑
- 不跨层导入（domain/ 导入 server/，data/ 导入 domain/）
- 不直接返回 dict 而不包装
- 不使用 `print` 输出日志（使用 `logging` 模块）
- 不在 __init__.py 之外暴露内部实现

```python
# src/domain/database/data.py
class DataDomain:
    async def create_documents(
        self, cname: str, documents: list[dict]
    ) -> list[str]:
        """批量创建文档，统一添加时间戳"""
        now = datetime.utcnow()
        for doc in documents:
            doc.setdefault("created_at", now)
            doc.setdefault("updated_at", now)

        result = await self._repo.insert_many(cname, documents)
        return [str(id) for id in result.inserted_ids]

# src/services/database/data_service.py
class DataService:
    async def create_documents(
        self, cname: str, documents: list[dict]
    ) -> dict[str, list[str]]:
        if not cname or not documents:
            raise ValidationException("cname and documents are required")
        if len(documents) > 100:
            raise ValidationException("一次最多创建 100 条文档")

        ids = await self._domain.create_documents(cname, documents)
        return {"inserted_ids": ids}
```

## 事务模式

### 跨集合操作

```python
# src/domain/bug/bug_domain.py
class BugDomain:
    def __init__(self, repository: Repository):
        self._repo = repository

    async def create_bug_with_notification(
        self, bug_data: dict, notify_users: list[str]
    ) -> str:
        """创建 Bug 并发送通知（非事务性，尽力而为）"""

        # 1. 创建 Bug（主操作）
        bug_id = await self._repo.insert("bugs", {
            **bug_data,
            "created_at": datetime.utcnow(),
            "status": "open",
        })

        # 2. 创建通知（尽力而为，失败不影响主操作）
        try:
            notifications = [
                {
                    "bug_id": bug_id,
                    "user_id": uid,
                    "type": "bug_created",
                    "read": False,
                    "created_at": datetime.utcnow(),
                }
                for uid in notify_users
            ]
            await self._repo.insert_many("notifications", notifications)
        except Exception as e:
            logger.warning(f"通知创建失败，Bug 已创建: {bug_id}", exc_info=True)

        return bug_id
```

### 关联数据一致性

```python
class DataDomain:
    async def create_with_relations(
        self, parent_cname: str, parent_data: dict,
        child_cname: str, children_data: list[dict]
    ) -> str:
        """创建父文档 + 关联子文档（非事务性，允许部分失败）"""

        # 1. 创建父文档
        now = datetime.utcnow()
        parent_data.setdefault("created_at", now)
        parent_id = await self._repo.insert(parent_cname, parent_data)

        # 2. 创建子文档（关联 parent_id）
        try:
            for child in children_data:
                child["parent_id"] = parent_id
                child.setdefault("created_at", now)
            await self._repo.insert_many(child_cname, children_data)
        except Exception as e:
            logger.error(f"子文档创建失败: parent_id={parent_id}", exc_info=True)
            # 不删除父文档，允许后续修复

        return parent_id
```

## 缓存模式

### 内存缓存（Domain 层）

```python
from functools import lru_cache
from datetime import datetime, timedelta

class KnowledgeDomain:
    def __init__(self, repository: Repository):
        self._repo = repository
        self._cache: dict[str, tuple[dict, datetime]] = {}
        self._cache_ttl = timedelta(seconds=60)

    async def get_file_info(self, file_path: str) -> dict | None:
        """获取文件信息（带 60s 缓存）"""

        # 检查缓存
        if file_path in self._cache:
            data, cached_at = self._cache[file_path]
            if datetime.utcnow() - cached_at < self._cache_ttl:
                return data

        # 缓存未命中，查询数据库
        doc = await self._repo.find_one("knowledge_files", {"file_path": file_path})
        if doc:
            self._cache[file_path] = (doc, datetime.utcnow())

        return doc

    def invalidate_cache(self, file_path: str | None = None):
        """清除缓存"""
        if file_path:
            self._cache.pop(file_path, None)
        else:
            self._cache.clear()
```

### 配置缓存

```python
class SettingsDomain:
```

## 校验模式

### Service 层参数校验

```python
class DataService:
    # 支持的集合白名单
    ALLOWED_COLLECTIONS = {
        "projects", "issues", "bugs", "sessions",
        "knowledge_files", "menus", "users", "labels",
    }

    async def query_documents(self, cname: str, **kwargs) -> dict:
        # 1. 集合白名单校验
        if cname not in self.ALLOWED_COLLECTIONS:
            raise ValidationException(f"不支持的集合: {cname}")

        # 2. 分页参数校验
        page_num = kwargs.get("page_num", 1)
        page_size = kwargs.get("page_size", 20)
        if page_num < 1:
            page_num = 1
        if page_size < 1 or page_size > 100:
            page_size = 20

        # 3. 排序字段校验
        order_by = kwargs.get("order_by")
        if order_by and not re.match(r'^[a-zA-Z_][a-zA-Z0-9_]*$', order_by):
            raise ValidationException(f"无效的排序字段: {order_by}")

        # 4. 委托 domain 层
        return await self._domain.query_documents(
            cname=cname,
            page_num=page_num,
            page_size=page_size,
            order_by=order_by,
            **{k: v for k, v in kwargs.items()
               if k not in ("page_num", "page_size", "order_by")},
        )
```

### Domain 层业务规则校验

```python
class BugDomain:
    VALID_SEVERITIES = {"critical", "major", "minor", "trivial"}
    VALID_STATUSES = {"open", "in_progress", "resolved", "closed", "reopened"}

    async def create_bug(self, data: dict) -> str:
        """创建 Bug — 业务规则校验"""

        # 1. 必填字段校验
        if not data.get("title"):
            raise ValidationException("Bug 标题不能为空")

        # 2. 枚举值校验
        severity = data.get("severity", "minor")
        if severity not in self.VALID_SEVERITIES:
            raise ValidationException(
                f"无效的严重程度: {severity}，有效值: {self.VALID_SEVERITIES}"
            )

        # 3. 状态转换校验
        status = data.get("status", "open")
        if status not in self.VALID_STATUSES:
            raise ValidationException(f"无效的状态: {status}")

        # 4. 关联数据校验
        if data.get("project"):
            project = await self._repo.find_one("projects", {"key": data["project"]})
            if not project:
                raise NotFoundException(f"项目不存在: {data['project']}")

        # 5. 创建 Bug
        data.setdefault("created_at", datetime.utcnow())
        data.setdefault("status", "open")
        return await self._repo.insert("bugs", data)
```

```python
        self._base_dir = base_dir

    async def read_file(self, target_file: str) -> str:
        """读取文件内容（安全路径校验）"""
        full_path = self._resolve_safe_path(target_file)

        if not os.path.exists(full_path):
            raise NotFoundException(f"文件不存在: {target_file}")

        async with aiofiles.open(full_path, "r") as f:
            return await f.read()

    async def write_file(self, target_file: str, content: str) -> None:
        """写入文件内容（安全路径校验 + 目录自动创建）"""
        full_path = self._resolve_safe_path(target_file)

        os.makedirs(os.path.dirname(full_path), exist_ok=True)

        async with aiofiles.open(full_path, "w") as f:
            await f.write(content)

    def _resolve_safe_path(self, target_file: str) -> str:
        """安全路径解析（防止目录穿越攻击）"""
        full_path = os.path.realpath(os.path.join(self._base_dir, target_file))
        base_real = os.path.realpath(self._base_dir)

        if not full_path.startswith(base_real):
            raise ValidationException(f"非法文件路径: {target_file}")

        return full_path
```

## 事件驱动模式

### 领域事件发布

```python
# src/domain/events.py
from dataclasses import dataclass, field
from datetime import datetime
from typing import Callable, Awaitable

@dataclass
class DomainEvent:
    name: str
    data: dict
    timestamp: datetime = field(default_factory=datetime.utcnow)

class EventBus:
    """简单的领域事件总线"""

    def __init__(self):
        self._handlers: dict[str, list[Callable[[DomainEvent], Awaitable[None]]]] = {}

    def subscribe(self, event_name: str, handler: Callable):
        if event_name not in self._handlers:
            self._handlers[event_name] = []
        self._handlers[event_name].append(handler)

    async def publish(self, event: DomainEvent):
        """发布事件（异步执行，不阻塞主流程）"""
        handlers = self._handlers.get(event.name, [])
        for handler in handlers:
            asyncio.create_task(self._safe_handle(handler, event))

    async def _safe_handle(self, handler, event: DomainEvent):
        try:
            await handler(event)
        except Exception as e:
            logger.error(f"Event handler failed: {event.name}", exc_info=True)

# 全局事件总线
event_bus = EventBus()
```
