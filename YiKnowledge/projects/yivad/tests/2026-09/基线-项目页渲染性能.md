### 3.1 npx lighthouse 4 条命令

```bash
# 1) 桌面模式 3 轮取中位数
npx lighthouse@12.0.0 http://127.0.0.1:3100/#/project \
  --preset=desktop \
  --only-categories=performance,best-practices \
  --output=json --output=html \
  --output-path=artifacts/lh_desktop_$(date +%F) \
  --chrome-flags="--headless=new" \
  --runs=3

# 2) 移动端模拟 + 网络节流
npx lighthouse http://127.0.0.1:3100/#/project \
  --preset=perf --throttling-method=simulate \
  --throttling.cpuSlowdownMultiplier=4 \
  --throttling.rttMs=150 --throttling.throughputKbps=1638 \
  --output=json --output-path=artifacts/lh_mobile_$(date +%F).json

# 3) 快照对比（对比两次基线快照）
npx @lhci/cli@0.14.x autorun \
  --collect.url=http://127.0.0.1:3100/#/project \
  --collect.numberOfRuns=3 \
  --assert.preset=lighthouse:recommended \
  --upload.target=temporary-public-storage \

   "largest-contentful-paint=error,1200" \
  --assert.assertions."cumulative-layout-shift=error,0.05"

  --project=chromium --reporter=json \
  --output=artifacts/playwright_trace_fps.zip \
  | tee artifacts/fps_report_$(date +%F).log
```

### 3.3 YiAi / YiPot RPC 端点接入片段

```ts
// src/services/projectsRpc.ts
import { createTRPCProxyClient, httpBatchLink } from '@trpc/client'

export const yiaiProjects = createTRPCProxyClient({
  links: [httpBatchLink({ url: import.meta.env.VITE_YIAI_RPC + '/trpc/projects' })],
})
export const yipotStats = createTRPCProxyClient({
  links: [httpBatchLink({ url: import.meta.env.VITE_YIPOT_RPC + '/trpc/stats' })],
})

// 加载 5 项目卡片数据
export async function loadProjectCards(limit = 5) {
  const [projects, stats] = await Promise.all([
    yiaiProjects.list.query({ limit, fields: ['id', 'name', 'logo', 'description'] }),
    yipotStats.summary.mutate({ projectIds: projects.map(p => p.id) }),
  ])
  return projects.map(p => ({ ...p, stats: stats[p.id] }))
}
```

## 五、回归脚本

```bash
#!/usr/bin/env bash
# ci/regression_project_render.sh
set -euo pipefail
OUT=artifacts/baseline_yivad_$(date +%F_%H%M)
mkdir -p "$OUT"

npx rsbuild build --mode production 2>&1 | tee "$OUT/rsbuild.log"
BUNDLE_MB=$(du -sm dist | cut -f1)
echo "bundle_unzip_mb=$BUNDLE_MB" >> "$OUT/metrics.env"

npx playwright start-server --command "npx rsbuild preview --port 3100" \
  --port 3100 --ready-on "curl -sSf http://127.0.0.1:3100/" \
  -- npx lighthouse http://127.0.0.1:3100/#/project \
    --preset=desktop --runs=3 --output=json --output-path="$OUT/lh.json"

npx playwright test tests/perf/project_scroll_fps.spec.ts \
  --project=chromium --reporter=json > "$OUT/fps.json"

node scripts/collect_metrics.js "$OUT" \
  | tee "$OUT/summary.json"

python scripts/compare_baseline.py --current "$OUT/summary.json" \
  --baseline snapshots/baseline_project_render_v20260928.json \
  --fail-on-regression lcp:1.10 cls:1.20 fps:0.92 bundle_mb:1.10
```

## 七、优化建议

1. **移动端 LCP 逼近 1.2s 红线：为 5 张项目卡片 logo 增加 `fetchpriority="high"` + 预加载关键字体 `preconnect` 两个 RPC 域名，预估 LCP 再降 0.12s。
2. **Rsbuild 体积接近 2MB**：将 `echarts` 与 `@trpc/client` 拆为独立 lazy chunk（项目页用不上仪表盘时不加载），可再减 0.22MB。
3. **FPS 低端机 P1% 掉帧**：卡片 hover 动效改用 `transform/opacity` 替换 `box-shadow/width` 触发重排，滚动 P1% 可回升至 54 帧以上。
